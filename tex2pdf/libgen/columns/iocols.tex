\newif\ifinsertskip

\def\beginiocolumnns{\begingroup\baselineskip=\the\baselineskip plus 1pt%
	\doublecolumnsetup \insertskipfalse}
\def\endiocolumns{\endgroup \pagegoal=\vsize}

\newbox\outcolumn
\newbox\incolumn
\long\def\colout#1\colin#2\colpar{%
	\begingroup%
	\setbox\outcolumn=\vbox{#1}%
	\setbox\incolumn=\vbox{#2}%
	\storecolumns%
	\buildpar%
	\endgroup%
}

\def\storecolumns{\directlua{libgen.iocols:store_lines_from("incolumn", "outcolumn")}}

\def\buildpar{\begingroup\count100=1%
	\loop%
	\getboxes%
	\ifnum\count100=1%
	\buildline\repeat%
	\endgroup}

\def\buildline{%
	\setbox2=\hbox to \textwidth{%
		\hbox to \hsize{\hfil\copy0}%
		\hfil\vrule\hfil%
		\hbox to \hsize{\hfil\copy1}}
	\ioaddline2
	\hbox attr 37=\linecount to \textwidth{%
		\hbox to \hsize{\hfil\box1}%
		\hfil\vrule\hfil%
		\hbox to \hsize{\hfil\box0}}}
\def\linecount{\directlua{tex.write(libgen.iocols.line_count)}}
\def\ioaddline#1{\directlua{libgen.iocols:add_line(#1)}}

\def\getboxes{\setbox0=\hbox{}\setbox1=\hbox{}
	\directlua{local next_item = libgen.iocols:get_next_item();
		if next_item == nil then
		tex.count[100] = 0;
		else
		local colin = next_item.i;
		local colout = next_item.o;
		if colin then tex.setbox(0, colin) end
		if colout then tex.setbox(1, colout) end
		end}}

\def\iocolsprocess{\directlua{libgen.iocols.pageprocess()}}
